SELECT working_sections.client_id, working_sections.name, count(completed_section_totals.task_id) FILTER (WHERE completed_section_totals.latest_closed_at >= %(latest_closed_at_1)s) AS sample_count, CASE WHEN (count(completed_section_totals.task_id) FILTER (WHERE completed_section_totals.latest_closed_at >= %(latest_closed_at_1)s) >= %(param_1)s) THEN CAST(round(percentile_cont(%(percentile_cont_1)s) WITHIN GROUP (ORDER BY completed_section_totals.group_seconds) FILTER (WHERE completed_section_totals.latest_closed_at >= %(latest_closed_at_1)s)) AS INTEGER) END AS typical_worker_seconds 
FROM working_sections LEFT OUTER JOIN (SELECT task_steps.working_section_id AS working_section_id, task_steps.task_id AS task_id, sum(task_steps.total_working_seconds) AS group_seconds, max(task_steps.closed_at) AS latest_closed_at 
FROM task_steps 
WHERE task_steps.workspace_id = %(workspace_id_1)s AND task_steps.state = %(state_1)s AND task_steps.is_deleted IS false AND task_steps.recorded_time_marked_wrong IS false GROUP BY task_steps.working_section_id, task_steps.task_id) AS completed_section_totals ON completed_section_totals.working_section_id = working_sections.client_id 
WHERE working_sections.workspace_id = %(workspace_id_2)s AND working_sections.is_deleted IS false GROUP BY working_sections.client_id, working_sections.name
